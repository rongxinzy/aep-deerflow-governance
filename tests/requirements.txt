# Test-only dependencies for the governance package's pytest suite.
#
# This repository ships no Python dependency manifest of its own: at runtime
# aep_deerflow_governance lives inside the deerflow-gov image, whose venv is
# built from the out-of-repo DeerFlow backend (DEVELOPMENT.md §4). Every
# `deerflow.*` import in the package is lazy and guarded
# (aep_deerflow_governance/__init__.py:235,263,646), so the suite needs only
# the langchain surface that the package and the tests import directly:
#
#   aep_deerflow_governance/__init__.py     langchain.agents.middleware.AgentMiddleware
#                                           langchain_core.messages.{SystemMessage,ToolMessage}
#   aep_deerflow_governance/email_tools.py  langchain_core.tools.{BaseTool,tool}
#   tests/test_knowledge_whitelist.py       langchain_core.messages.ToolMessage
#
# Floors mirror the runtime pins in the DeerFlow backend
# (packages/harness/pyproject.toml: "langchain>=1.3";
# backend/pyproject.toml dev group: "pytest>=9.0.3"). langchain-core arrives
# transitively via langchain but is declared because we import it by name.
# Run from the repository root with `python -m pytest tests/`, which puts the
# root on sys.path (bare `pytest` cannot import the package).
pytest>=9.0.3
langchain>=1.3
langchain-core>=1.3
